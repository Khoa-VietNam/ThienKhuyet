# Small, repeatable spirit-stone shop; choices are disabled when the player cannot afford them.
dialogue merchant_shop
  node start speaker=merchant_trinh text=dlg.merchant.1 expr=smile
    choice text=dlg.merchant.buy_paste goto=receipt_paste if="item:spirit_stone>=8" do="take:spirit_stone*8;give:healing_paste*1"
    choice text=dlg.merchant.buy_pill goto=receipt_pill if="item:spirit_stone>=18" do="take:spirit_stone*18;give:qi_pill_low*1"
    choice text=dlg.merchant.leave goto=end
  node receipt_paste speaker=merchant_trinh text=dlg.merchant.receipt_paste expr=smile next=end
  node receipt_pill speaker=merchant_trinh text=dlg.merchant.receipt_pill expr=smile next=end
